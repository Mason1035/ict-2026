cmake_minimum_required(VERSION 3.16)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(COMPONENTS main)
set(PROJECT_VER "0.1.0-framework")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(zhishao_node01)
if(NOT IDF_VERSION_MAJOR EQUAL 5 OR NOT IDF_VERSION_MINOR EQUAL 5)
    message(FATAL_ERROR "NODE-01 requires ESP-IDF 5.5.x; do not upgrade automatically")
endif()
if(NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "NODE-01 requires esp32s3")
endif()
# IDF's directory flags otherwise append a later default C++ standard.
foreach(component node_board node_contracts node_logic node_runtime main)
    idf_component_get_property(lib ${component} COMPONENT_LIB)
    target_compile_options(${lib} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-std=c++17>")
endforeach()
